budgetctl — cheatsheet
# Import bank CSV
$ budgetctl import --file n26-export.csv
$ budgetctl import --file ing-2026-07.csv --bank ing

# Monthly summary
$ budgetctl summary --month 2026-07

# List transactions
$ budgetctl list --month 2026-07
$ budgetctl list --category food

# Set budget goals
$ budgetctl goal set food 500
$ budgetctl goal list

# Categorize a transaction
$ budgetctl categorize --id 42 --category dining

# Add a category rule
$ budgetctl rule add "Spotify" entertainment

# Detect recurring payments
$ budgetctl recurring

# Start MCP server
$ budgetctl mcp
