taskctl — cheatsheet
# Show today's tasks
$ taskctl today

# TUI task manager
$ taskctl tui

# Add a task
$ taskctl add "Review PR #42" --due today
$ taskctl add "Write docs" --due 2026-07-10 --priority high

# Complete tasks
$ taskctl done --id 1
$ taskctl done --all-today

# Pomodoro timer
$ taskctl pomodoro
$ taskctl pomodoro --task 1

# Sync with Apple Reminders
$ taskctl sync

# Start daemon
$ taskctl daemon start

# Start MCP server
$ taskctl mcp
